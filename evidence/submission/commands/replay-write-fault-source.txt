import argparse
import json
from pathlib import Path
from computeruse.capability import load_capability
from computeruse.profile import load_profile
from evaluation import integrated as evaluation
from evaluation.operator import FileChannel

parser=argparse.ArgumentParser()
parser.add_argument('--site',required=True)
parser.add_argument('--capability',required=True,type=Path)
parser.add_argument('--out',required=True,type=Path)
args=parser.parse_args()
app={'responsive':'white-label-responsive','components':'web-component-operations','canvas':'canvas-teller'}[args.site]
args.out.mkdir(parents=True,exist_ok=False)
evaluation.CHANNEL=FileChannel(args.out/'operator')
profile=load_profile(evaluation.profile_path(app))
capability=load_capability(args.capability)
task=next(t for t in evaluation.TASKS if t.name=='open_account')
case=task.faulted
injection=evaluation.FAULTS[(app,task.name)][0]
result=evaluation.replay_case(app,f'http://127.0.0.1:{evaluation.APPS[app]}/',args.out,profile,capability,task,case,injection)
(args.out/'summary.json').write_text(json.dumps(result,indent=2)+'\n')
print(json.dumps(result),flush=True)
